@startuml
!include https://raw.githubusercontent.com/plantuml-stdlib/C4-PlantUML/master/C4_Container.puml

LAYOUT_LANDSCAPE()

title Diagrama de Containers

Person(visitante, "Visitante", "Usuário não autenticado que consome a plataforma")
Person(aluno, "Aluno", "Aluno que pode participar dos projetos")
Person(professor, "Professor", "Professor que pode criar dos projetos")
Person(comunidade, "Comunidade Externa", "usuário autenticado como comunidade externa que interage melhor na plataforma")


System_Boundary(system, "TADSXANDRIA BIBLIOTECA") {
  Container(ini, "Visualizar projetos", "Pensar", "Permite a visualização dos projetos")

  Container(proj, "Gerência de projetos", "Pensar", "Permite a gerência de projetos")
  Container(proj_back, "Projeto Cadastro", "Pensar", "Usa a API do usuário para manuseio de dados")

  Container(ini_back, "Serviço de listagem", "Pensar", "Processa os projetos cadastrados mais avaliados")
  ContainerDb(database, "Banco de Dados", "Pensar", "Armazena dados dos projetos")
}

System_Ext(suap, "SUAP", "Auxilia a vinculação dos dados com o sistema, através da matrícula do SUAP")
System_Ext(github, "GitHub", "Auxilia a vinculação dos dados com o sistema, através da conta GitHub")
System_Ext(google, "Google", "Auxilia a vinculação dos dados com o sistema, através da conta Google")

Rel(visitante, ini, "Visita a plataforma para conhecê-la")
Rel(aluno, ini, "Interage nos projetos cadastrados")
Rel(professor, ini, "Interage nos projetos cadastrados")
Rel(comunidade, ini, "Interage nos projetos cadastrados")

Rel(aluno, proj, "Edita os projetos que participa, sendo como orientado")
Rel(professor, proj, "Gerencia os projetos que o cadastrou, sendo como orientador")

Rel(proj, proj_back, "Faz a chamado do cadastro")
Rel(proj_back, database, "Lê e escrever os dados para o banco")



Rel(ini, ini_back, "Chamada para o processo de listagem")
Rel(ini_back, database, "Acessa informações dos projetos")

@enduml
